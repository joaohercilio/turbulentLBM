cmake_minimum_required(VERSION 3.20)
project(ribsVideo LANGUAGES C CUDA)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_ARCHITECTURES 89)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
find_package(HDF5 REQUIRED COMPONENTS C)
add_executable(ribsVideo main.cpp)
set_source_files_properties(main.cpp PROPERTIES LANGUAGE CUDA)
# copias locais de core/ e lattice/ (lattice.hpp com d3q27_3rd), o repositorio fica intocado
target_include_directories(ribsVideo PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(ribsVideo PRIVATE DEVICE=1)
target_link_libraries(ribsVideo PRIVATE HDF5::HDF5)
